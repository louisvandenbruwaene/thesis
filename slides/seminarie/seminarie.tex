% ---- 1. simple graph, undirected, edge -------------------------------------
\vframe{Simple graph \textperiodcentered\ edge}{\ell_m(n)}
{$\floorfrac{m(n-1)}{2}\;=\;9$ \ \sProved\ \ {\scriptsize\color{black!50}$n\ge m$}}
{$\floorfrac{m(n-1)}{2}\;=\;9$ \ \sProved}
{\begin{vsteps}
  \vstep{a Gomory--Hu tree summarises every pairwise connectivity in $n-1$ cuts}
  \vstep{each cut has capacity at most $m-1$, so together they carry $(m-1)(n-1)$}
  \vstep{every edge is charged to two of those cuts, which halves the count}
  \vstep{a star with a degree-$(m-2)$ spoke graph on its leaves attains it}
\end{vsteps}}

% ---- 2. simple graph, undirected, vertex -----------------------------------
\vframe{Simple graph \textperiodcentered\ vertex}{k_m(n)}
{$k_m(n)=\ell_m(n)\;=\;9$ \ \sProved\ \ {\scriptsize\color{black!50}$m\le4$}}
{$\floorfrac{m(n-1)}{2}\;=\;9$ \ \sProved}

% ---- 3. simple digraph, directed, arc --------------------------------------
\vframe{Simple digraph \textperiodcentered\ arc}{\ell_m^{\dir}(n)}

{\begin{vsteps}
  \vstep{a reachability skeleton is a subdigraph preserving who reaches whom}

  \vstep{a skeleton carries at most $M(n)$ arcs, which bounds the digraph}

         one-way bipartite digraph each attain it}
\end{vsteps}}

% ---- 6. multigraph, undirected, vertex -------------------------------------

\end{vsteps}}

% ---- 9. simple hypergraph, undirected, edge --------------------------------
\vframe{Simple hypergraph \textperiodcentered\ edge}{\ell_m^{(r)}(n)}
{$\floorfrac{(m-1)(n-1)}{r-1}\;=\;4$ \ \sProved}
{$\floorfrac{(m-1)(n-1)}{r-1}\;=\;4$ \ \sProved\ \ {\scriptsize\color{black!50}$m-1\le\binom{n-2}{r-2}$}}
{\begin{vsteps}
  \vstep{induction on a single hyperedge cut, with no tree anywhere in it}

  \vstep{the reduction to graphs points the wrong way: replacing a hyperedge by

\end{vsteps}}

% ---- 10. simple hypergraph, undirected, vertex -----------------------------
\vframe{Simple hypergraph \textperiodcentered\ vertex}{k_m^{(r)}(n)}
{$\floorfrac{n-1}{r-1}\;=\;4$ \ \sProved\ \ {\scriptsize\color{black!50}$m=2$}}
{$\floorfrac{n-1}{r-1}\;=\;4$ \ \sProved}
{\begin{vsteps}
  \vstep{at $m=2$ feasibility says exactly that the incidence graph is a forest,
         since a cycle in it would already be two disjoint routes}
  \vstep{a forest on the $n$ vertices together with the $|E|$ hyperedges has at
         most $n+|E|-1$ edges}
  \vstep{each hyperedge contributes $r$ of them, so $r|E|\le n+|E|-1$, which
         rearranges to $|E|(r-1)\le n-1$}
  \vstep{the star hypertree's incidence graph is a tree, so it attains it}
\end{vsteps}}

% ---- 13. directed simple hypergraph, arc -----------------------------------
\vframe{Directed simple hypergraph \textperiodcentered\ arc}{\ell_m^{\dir\,(r)}(n)}
{$8$ \ \sProved\ \ {\scriptsize\color{black!50}enumeration at this size}}
{$n\cdot\min\bigl(m-1,\tbinom{n-1}{r-1}\bigr)\;=\;8$ \ \sProved}
{\begin{vsteps}
  \vstep{every route out of a vertex has to start with one of that vertex's own
