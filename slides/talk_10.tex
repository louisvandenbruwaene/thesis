\begin{frame}{The question}
\vfill
\centering
{\large How many edges can a graph on $n$ vertices carry}\\[2pt]
{\large if \emph{no} pair of vertices has $m$ edge-disjoint paths?}

\vspace{16pt}
{\color{KULblauw3b}\Large $\displaystyle \ell_m(n)=\max\bigl\{\,|E(G)| \;:\; |V(G)|=n,\ \lmax_G\le m-1\,\bigr\}$}

\vspace{18pt}
\begin{minipage}{0.62\textwidth}
\centering
{\color{black!45}\small Mader settled it for simple graphs, $n\ge m$:}\\[6pt]
{\color{KULblauw1}\Large $\displaystyle \ell_m(n)=\floorfrac{m(n-1)}{2}$}
\end{minipage}

\vspace{18pt}
{\footnotesize\color{black!60} This thesis moves that question along three axes: model, direction, separation. Twelve variants.}
\vfill
\end{frame}

% --- 9 -----------------------------------------------------------------------
\begin{frame}{What is new here}
\vfill
\begin{itemize}\setlength{\itemsep}{14pt}
  \item[\color{KULblauw1}\textbf{1}] \textbf{Hypergraph vertex problem: settled at $m=2$; sharp at $m=3$ in its construction range}\\
        {\small\color{black!60} a universal $m=3$ upper bound follows from a new incidence-graph rank bound; equality is proved for $r\ge3$ when $2\le\binom{n-2}{r-2}$}
  \item[\color{KULblauw1}\textbf{2}] \textbf{One-way simple graphs: an unconditional quadratic bound}\\
        {\small\color{black!60} $|A| \le \floorfrac{n^2}{4} + 4(m-1)(n-1)$, so the leading constant and the order of the next term are theorems, not conjectures}
  \item[\color{KULblauw1}\textbf{3}] \textbf{One-way multigraphs, solved for every $n$ and every $m$}\\
        {\small\color{black!60} $L_m^{\mathrm{dir}}(n)=(m-1)\max\bigl(2(n-1),\lfloor n^2/4\rfloor\bigr)$, by deleting a sparse subgraph that alone remembers who can reach whom, once per unit of $m-1$}
\end{itemize}
\vfill
\end{frame}
