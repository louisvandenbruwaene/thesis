\begin{frame}{The question}
\vfill
\centering
{\large How many edges can a graph on $n$ vertices carry}\\[2pt]
{\large if \emph{no} pair of vertices has $m$ edge-disjoint paths?}

\vspace{16pt}
{\color{KULblauw3b}\Large $\displaystyle \ell_m(n)=\max\bigl\{\,|E(G)| \;:\; |V(G)|=n,\ \lmax_G\le m-1\,\bigr\}$}

\vspace{18pt}
\begin{minipage}{0.62\textwidth}
\centering
{\color{black!45}\small Mader, for simple graphs and all $n\ge2$, $m\ge2$:}\\[6pt]
{\color{KULblauw1}\Large $\displaystyle \ell_m(n)\le\floorfrac{m(n-1)}{2}$}
\end{minipage}

\vspace{18pt}
{\footnotesize\color{black!60} Proved here again from the Gomory--Hu tree, and a construction meets it for every $n\ge m$. Then three toggles on the model and one on the separation turn one question into sixteen.}
\vfill
\end{frame}

\begin{frame}{What is new here}
\vfill
\begin{itemize}\setlength{\itemsep}{13pt}
  \item[\color{KULblauw1}\textbf{1}] \textbf{One-way multigraphs, solved for every $n$ and every $m$}\\
        {\small\color{black!60} $L_m^{\dir}(n)=(m-1)\max\bigl(2(n-1),\floor{n^2/4}\bigr)$, by deleting a sparse reachability-preserving subgraph once per unit of $m-1$ rather than a vertex at a time. The quadratic extremisers are classified for $n\ge\max(8,2m)$.}
  \item[\color{KULblauw1}\textbf{2}] \textbf{One-way simple graphs: an unconditional quadratic bound}\\
        {\small\color{black!60} $|A(D)|\le\floorfrac{n^2}{4}+4(m-1)(n-1)$, with no hypothesis on the shape of the extremiser, so the leading constant and the order of the next term are theorems. The same count re-runs under the stricter separation.}
  \item[\color{KULblauw1}\textbf{3}] \textbf{Hyperedge networks: the vertex problem at $m=2$ and $m=3$, and a directed leading term}\\
        {\small\color{black!60} $\floor{2(n-1)/(r-1)}$ is a universal $m=3$ bound from a new incidence-graph rank lemma, attained for $r\ge3$ when $2\le\binom{n-2}{r-2}$. One-way, the leading term is $(m-1)n^2/\bigl(4(r-1)\bigr)$ under both separations.}
\end{itemize}
\vfill
\end{frame}

\begin{frame}{What is left}
\vfill
\begin{itemize}\setlength{\itemsep}{12pt}
  \item[{\color{stPartial}$\blacksquare$}] The undirected \emph{vertex} problem at $m\ge6$, open since 1974
  \item[{\color{stConj}$\blacksquare$}] The decomposition that would turn the directed conjecture into a theorem: that every extremal non-hub digraph splits as $A\cup B$ with no arc coming back
  \item[{\color{stConj}$\blacksquare$}] Whether the directed vertex value equals the arc value at $m\ge3$, which Whitney's inequality does not hand over
  \item[{\color{stOpen}$\blacksquare$}] The exact values behind the leading terms now settled, one-way and for hyperedges alike
\end{itemize}
