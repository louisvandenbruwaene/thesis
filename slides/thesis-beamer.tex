% \figframe{frame title}{figure file}{one-line reading}
\newcommand{\figframe}[3]{%
  \begin{frame}{#1}
    \vfill
    \centering
    \includegraphics[width=\textwidth,height=0.66\textheight,keepaspectratio]{#2}

    \vspace{3pt}
    {\footnotesize\color{black!60} #3}
    \vfill
  \end{frame}%
}

% \figframetall{...}: for the multi-panel grids, which are close to square and
% need every millimetre of height to stay legible.  Its third argument must be
% a SHORT single line.
\newcommand{\figframetall}[3]{%
  \begin{frame}{#1}
    \vspace{-2pt}
    \centering
    \includegraphics[width=\textwidth,height=0.75\textheight,keepaspectratio]{#2}

    \vspace{1pt}
    {\scriptsize\color{black!60} #3}
  \end{frame}%
}

% \figframewide{...}: for the very wide panels (ratio above 2), which need the
% full width and no height cap to stay legible.
\newcommand{\figframewide}[3]{%
  \begin{frame}{#1}
    \vfill
    \centering
    \adjustbox{max totalheight=0.66\textheight}{\includegraphics[width=\textwidth]{#2}}

    \vspace{6pt}
    {\footnotesize\color{black!60} #3}
    \vfill
  \end{frame}%
}

% \tikzframe{title}{\figmacro}{reading line}: the same layout for a TikZ
% picture from thesis-figures.tex.  The picture is blown up to the text width
% and then capped in height, so a wide one fills the slide and a tall one is
% shrunk to fit rather than running off the bottom.
\newcommand{\tikzframe}[3]{\tikzframescaled{#1}{#2}{#3}{1.0}}

% \tikzframescaled{...}{fraction}: same, for a picture that would look absurd
% stretched to the full text width.
\newcommand{\tikzframescaled}[4]{%
  \begin{frame}{#1}
    \vfill
    \centering
    \adjustbox{max totalheight=0.64\textheight}{\resizebox{#4\textwidth}{!}{#2}}

    \vspace{8pt}
    {\footnotesize\color{black!60} #3}
    \vfill
  \end{frame}%
}

% ============================================================================
%  THE SIXTEEN-VARIANT TABLE
%  The content of tab:summary in chapters/ch3_synthesis.tex, set for a slide:
%  the cross references are replaced by short words and a status chip is added.
%  It must be kept in step with tab:summary by hand.  The chips are the four
%  regimes of fig:variant-tree-status: PROVED (closed form for every n and m),
%  RANGE (proved for a range of m, open beyond it), CONJ (conjectured formula
%  with a proved lower bound), and LEADING TERM (the n^2 term is a theorem, the
%  exact value is not).
% ============================================================================
\newcommand{\sixteentable}{%
\renewcommand{\arraystretch}{1.10}
\scriptsize
\resizebox{\textwidth}{!}{%
\begin{tabular}{@{}lp{0.40\textwidth}p{0.40\textwidth}@{}}
\toprule
Model & Edge or arc-disjoint & Vertex-disjoint \\
\midrule
\multicolumn{3}{@{}l}{\color{KULblauw1}\bfseries Undirected} \\
simple graph
  & $\floorfrac{m(n-1)}{2}$ \ \chip{stProved}{PROVED} Mader
  & the edge value for $m\le4$, and $\floor{8n/3}-4$ at $m=5$ \ \chip{stPartial}{RANGE} open $m\ge6$ \\
multigraph
  & $(m-1)(n-1)$ \ \chip{stProved}{PROVED}
  & the simple value \ \chip{stPartial}{RANGE} parallel copies do not raise $\kap$ \\
$r$-hypergraph
  & $\le\floorfrac{(m-1)(n-1)}{r-1}$, equality in the stated range \ \chip{stProved}{PROVED}
  & $\floorfrac{n-1}{r-1}$ at $m=2$, $\le\floorfrac{2(n-1)}{r-1}$ at $m=3$ \ \chip{stPartial}{RANGE} open $m\ge4$ \\
$r$-multi-hyper
  & the same bound, equality when $(r-1)\mid(n-1)$ \ \chip{stProved}{PROVED}
  & the same bounds, repeats closing cases at $m=3$ \ \chip{stPartial}{RANGE} \\
\addlinespace[2pt]
\midrule
\multicolumn{3}{@{}l}{\color{KULblauw1}\bfseries Directed} \\
simple digraph
  & $M(n)$ at $m=2$; $n^2/4+\Theta_m(n)$ for $m\ge3$ \ \chip{stProved}{PROVED} $m=2$ \ \chip{stConj}{CONJ} $m\ge3$
  & $M(n)$ at $m=2$; $n^2/4+\Theta_m(n)$ for $m\ge3$ \ \chip{stProved}{PROVED} $m=2$ \ \chip{stConj}{CONJ} $m\ge3$ \\
multigraph
  & $(m-1)M(n)$ \ \chip{stProved}{PROVED} every $n$ and $m$
  & the simple digraph value \ \chip{stConj}{CONJ} $m\ge3$ \\
$r$-hypergraph
  & $(1+o(1))\dfrac{(m-1)n^2}{4(r-1)}$ \ \chip{stOpen}{LEADING TERM}
  & the same leading term \ \chip{stOpen}{LEADING TERM} \\
$r$-multi-hyper
  & the same leading term, repeats allowed \ \chip{stOpen}{LEADING TERM}
  & the same leading term \ \chip{stOpen}{LEADING TERM} \\
\bottomrule
\end{tabular}%
}

\vspace{5pt}
{\tiny\color{black!55} $M(n)=\max\bigl(2(n-1),\floor{n^2/4}\bigr)$, and the simple-graph formulas assume $n\ge m$. \
\chip{stProved}{PROVED} closed form for every $n$ and $m$. \
\chip{stPartial}{RANGE} proved for a range of $m$, open beyond it. \
\chip{stConj}{CONJ} conjectured formula, lower bound proved and the small cases exhausted. \
\chip{stOpen}{LEADING TERM} the $n^2$ term is a theorem, the exact value is not.}%
}
