%  Both decks are figures and tables with a title and one line underneath, so

% \figframe{frame title}{figure file}{one-line reading}
\newcommand{\figframe}[3]{%
  \begin{frame}{#1}
    \vfill
    \centering
    \includegraphics[width=\textwidth,height=0.70\textheight,keepaspectratio]{#2}

    \vspace{3pt}
    {\footnotesize\color{black!60} #3}
    \vfill
  \end{frame}%
}

% \figframetall{...}: for the multi-panel grids (twelve small panels, or the
% gallery), which are close to square and need every millimetre of height to
% stay legible.  Its third argument must be a SHORT single line.
\newcommand{\figframetall}[3]{%
  \begin{frame}{#1}
    \vspace{-2pt}
    \centering
    \includegraphics[width=\textwidth,height=0.80\textheight,keepaspectratio]{#2}

    \vspace{1pt}
    {\scriptsize\color{black!60} #3}
  \end{frame}%
}

% \figframewide{...}: for the very wide panels (ratio above 2), which need the
% full width and no height cap to stay legible.
\newcommand{\figframewide}[3]{%
  \begin{frame}{#1}
    \vfill
    \centering
    \includegraphics[width=\textwidth]{#2}

    \vspace{6pt}
    {\footnotesize\color{black!60} #3}
    \vfill
  \end{frame}%
}
