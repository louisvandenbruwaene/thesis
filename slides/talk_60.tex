\sectionframe{Part 1}{The question}

\begin{frame}{\Erdos{} Problem 915}
\vfill
\centering
{\large How many edges can a graph on $n$ vertices carry}\\[2pt]
{\large if \emph{no} pair of vertices has $m$ edge-disjoint paths?}

\vspace{14pt}
{\color{KULblauw3b}\Large $\displaystyle \ell_m(n)=\max\bigl\{\,|E(G)| \;:\; |V(G)|=n,\ \lmax_G\le m-1\,\bigr\}$}

\vspace{16pt}
\begin{minipage}{0.66\textwidth}
\centering
{\color{black!45}\small Mader, for simple graphs and $n\ge m$:}\\[6pt]
{\color{KULblauw1}\Large $\displaystyle \ell_m(n)=\floorfrac{m(n-1)}{2}$}
\end{minipage}

\vspace{16pt}
{\footnotesize\color{black!60} Posed by Bollob\'{a}s and \Erdos. Menger turns ``many routes'' into ``costly cut'', and the whole thesis is built on that one exchange.}
\vfill
\end{frame}

\sectionframe{Part 3}{What falls to a hand proof}

\begin{frame}{The short proofs}
\vfill
\centering
\renewcommand{\arraystretch}{1.7}
\begin{tabular}{@{}ll@{}}
\toprule
Multigraphs, edge-disjoint & $L_m(n)=(m-1)(n-1)$ \\
Simple graphs, vertex-disjoint at $m=2$ & $k_2(n)=n-1$, exactly the trees \\
Simple graphs, $m\le4$, $n\ge m$ & $k_m(n)=\ell_m(n)=\floorfrac{m(n-1)}{2}$ \quad {\scriptsize (Leonard)} \\
Hypergraphs, edge-disjoint & $\ell_m^{(r)}(n)=\floorfrac{(m-1)(n-1)}{r-1}$ \\
Random graphs & threshold $p^{*}=m/n$ \\
\bottomrule
\end{tabular}

\vspace{18pt}
{\footnotesize\color{black!60} One argument serves the first and the fourth: charge every edge to the Gomory--Hu tree, a summary of all $\binom{n}{2}$ connectivities in $n-1$ numbers.}
\vfill
\end{frame}

\figframe{Linear king against quadratic cabinet}
  {directed_crossover.png}

\begin{frame}{The directed conjecture, and what is unconditional}
\vfill
\centering
{\color{black!45}\small Conjectured value, lower bound proved for all $m$:}\\[6pt]
{\color{stConj}\large $\displaystyle \ell_m^{\mathrm{dir}}(n)=\max\Bigl(m(n-1),\ \floorfrac{(n+m-2)^2}{4}\Bigr)$}

\vspace{20pt}
{\color{black!45}\small Proved here with no hypothesis on the extremal shape:}\\[6pt]
{\color{KULblauw1}\large $\displaystyle |A(D)| \;\le\; \floorfrac{n^2}{4} \;+\; 4(m-1)(n-1)$}\\[8pt]
{\color{black!45}\small and therefore}\\[6pt]
{\color{KULblauw1}\large $\displaystyle \ell_m^{\mathrm{dir}}(n)=\frac{n^2}{4}+\Theta_m(n)\qquad(m\ge3)$}

\vspace{16pt}
\begin{minipage}{0.86\textwidth}
{\footnotesize\color{black!60} Leading constant and order of the second term: both now theorems. What is left is one coefficient, $(m-2)/2$ conjectured against $4(m-1)$ proved.}
\end{minipage}
\vfill
\end{frame}

\figframewide{Enumeration detonates}

\begin{frame}{Three jobs, kept apart}
\vfill
\begin{columns}[T]
\begin{column}{0.32\textwidth}
  {\color{KULblauw1}\bfseries\large MEASURE}\\[8pt]
  {\small One integer maximum flow, on one multiplicity matrix, for all twelve variants.}\\[8pt]
  {\scriptsize\color{black!55} What it reports about a given graph is a fact.}
\end{column}
\begin{column}{0.32\textwidth}
  {\color{stProved}\bfseries\large PROVE}\\[8pt]
  {\small Exhaust every graph of a fixed size, or run the cut-counting program.}\\[8pt]
  {\scriptsize\color{black!55} A finished sweep is an upper bound with the standing of a hand proof.}
\end{column}
\begin{column}{0.32\textwidth}
  {\color{stOpen}\bfseries\large DISCOVER}\\[8pt]
  {\small Cool a random graph, or climb greedily with a tabu list.}\\[8pt]
  {\scriptsize\color{black!55} Only ever exhibits a graph. Lower bounds only. Never a proof.}
\end{column}
\end{columns}

\vspace{24pt}
\centering
{\color{black!55}\small The honesty contract: the search proposes a lower bound, a proof or a finished exhaustion disposes the upper bound,}\\
{\color{black!55}\small and only when the two meet is a value called settled.}
\vfill
\end{frame}

\begin{frame}{What is new here}
\vfill
\begin{itemize}\setlength{\itemsep}{5pt}
  \item[\color{KULblauw1}\textbf{1}] \textbf{The hypergraph vertex problem: settled at $m=2$; sharp at $m=3$ in its construction range}\\
        {\footnotesize\color{black!60} a universal $m=3$ upper bound follows from a new incidence-graph rank bound. Equality is proved for $r\ge3$ when $2\le\binom{n-2}{r-2}$.}
  \item[\color{KULblauw1}\textbf{2}] \textbf{One-way simple graphs: both separations are $n^2/4+\Theta_m(n)$ for fixed $m\ge3$}\\
        {\footnotesize\color{black!60} no assumption about the extremal shape, which every earlier route to this bound needed. The two statements are separate: the vertex separation is not a corollary of the arc one. The exact linear coefficient remains open.}
  \item[\color{KULblauw1}\textbf{3}] \textbf{One-way multigraphs, solved outright: $L_m^{\mathrm{dir}}(n)=(m-1)\max\bigl(2(n-1),\floorfrac{n^2}{4}\bigr)$}\\
        {\footnotesize\color{black!60} every $n$ and every $m$, by peeling off one sparse reachability-preserving subgraph per unit of $m-1$. No parity split, no auxiliary conjecture, and the finite $n=7$ fact this problem once needed a solver for now follows in three lines.}
  \item[\color{KULblauw1}\textbf{4}] \textbf{One-way hyperedge networks: the leading term settled}\\
        {\footnotesize\color{black!60} $(1+o(1))(m-1)n^2/\bigl(4(r-1)\bigr)$, for edge- and vertex-disjoint routes alike, so the bipartite construction is asymptotically optimal.}
\end{itemize}
\vfill
\end{frame}

\begin{frame}{What is left}
\vfill
\begin{itemize}\setlength{\itemsep}{11pt}
  \item {\color{stOpen}$\blacksquare$} \textbf{The undirected vertex problem at $m\ge6$}, open since 1974, extremal structure genuinely unknown
  \item {\color{stConj}$\blacksquare$} \textbf{The missing decomposition}: that an extremal non-hub digraph splits as $A\cup B$ with every arc from $A$ to $B$, none inside $A$, none coming back. Four hypotheses, of which the backward arc is the hardest.
  \item {\color{stConj}$\blacksquare$} \textbf{The directed vertex problem at $m\ge3$}, which Whitney's inequality pointedly does \emph{not} hand over with the arc case. Its leading term is settled all the same, by borrowing the route family rather than the bound
  \item {\color{stConj}$\blacksquare$} \textbf{The exact values} behind the settled leading terms: the linear coefficient one-way, and the hyperedge value at finite $n$
\end{itemize}

\vspace{18pt}
\centering
{\color{black!55}\small The map is filled in where honest mathematics could fill it,}\\
{\color{black!55}\small and where it could not, the blanks are marked precisely.}
\vfill
\end{frame}
