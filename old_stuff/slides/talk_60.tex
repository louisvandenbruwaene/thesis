\sectionframe{Part 1}{The question}

\begin{frame}{\Erdos{} Problem 915}
\vfill
\centering
{\large How many edges can a graph on $n$ vertices carry}\\[2pt]
{\large if \emph{no} pair of vertices has $m$ edge-disjoint paths?}

\vspace{14pt}
{\color{KULblauw3b}\Large $\displaystyle \ell_m(n)=\max\bigl\{\,|E(G)| \;:\; |V(G)|=n,\ \lmax_G\le m-1\,\bigr\}$}

\vspace{16pt}
\begin{minipage}{0.66\textwidth}
\centering
{\color{black!45}\small Mader, for all $n\ge2$ and $m\ge2$:}\\[6pt]
{\color{KULblauw1}\Large $\displaystyle \ell_m(n)\le\floorfrac{m(n-1)}{2}$}
\end{minipage}

\vspace{16pt}
{\footnotesize\color{black!60} Posed by Bollob\'{a}s and \Erdos. The ceiling $\lmax_G\le m-1$ says every pair can be severed by fewer than $m$ edges, so Menger turns ``many routes'' into ``costly cut'', and the whole thesis is built on that one exchange.}
\vfill
\end{frame}

\tikzframesmall{Menger, on a five-vertex example}
  {\figmenger}
  {Three edge-disjoint $1$--$3$ routes meet the minimal three-edge cut around vertex $1$, so $\lec(1,3)=3$. A route count is a cut count.}
  {0.42}

\begin{frame}{The one tool: the Gomory--Hu tree}
\vfill
\centering
\begin{minipage}{0.82\textwidth}
{\large Every graph has a weighted spanning tree in which $\lec_G(u,v)$ is the \emph{lightest} edge on the tree path from $u$ to $v$.}
\end{minipage}

\vspace{20pt}
{\color{KULblauw3b}\large All $\binom{n}{2}$ local connectivities, stored in $n-1$ numbers.}\\[6pt]
{\color{KULblauw1}\large $\lmax_G$ is simply the heaviest edge of that tree.}

\vspace{22pt}
\begin{minipage}{0.86\textwidth}
{\footnotesize\color{black!60} A double count on that tree proves Mader's bound, independently of Mader, and the same count returns for the multigraph value and for the hypergraph bound one edge size up. Digraphs have no Gomory--Hu tree, which is exactly where the hand proofs start to fail.}
\end{minipage}
\vfill
\end{frame}

\sectionframe{Part 3}{What falls to a hand proof}

\begin{frame}{The short proofs}
\vfill
\centering
\renewcommand{\arraystretch}{1.7}
\small
\adjustbox{max width=\textwidth}{%
\begin{tabular}{@{}ll@{}}
\toprule
Simple graphs, edge-disjoint & $\ell_m(n)\le\floorfrac{m(n-1)}{2}$ \quad {\scriptsize (Mader, re-proved here, and attained for $n\ge m$)} \\
Multigraphs, edge-disjoint & $L_m(n)=(m-1)(n-1)$ \\
Simple graphs, $m\le4$, $n\ge m$ & $k_m(n)=\ell_m(n)=\floorfrac{m(n-1)}{2}$ \quad {\scriptsize (B\'{a}rtfai, Bollob\'{a}s, Leonard)} \\
Hypergraphs, edge-disjoint & $\le\floorfrac{(m-1)(n-1)}{r-1}$, attained in a stated range \\
Hypergraphs, vertex-disjoint at $m=2$ & $K_2^{(r)}(n)=\floorfrac{n-1}{r-1}$, and the star hypertree is simple \\
\bottomrule
\end{tabular}}

\vspace{18pt}
{\footnotesize\color{black!60} One argument serves the first, the second and the fourth: charge every edge to the Gomory--Hu tree.}
\vfill
\end{frame}

\figframe{The first divergence, at $m=5$}
  {edge_vertex_divergence.png}
  {Edge and vertex agree through $m=4$. S{\o}rensen and Thomassen give $k_5(n)=\floor{8n/3}-4$ for $n\ge6$, $n\ne7,12$, which first passes the edge value at $n=14$. Beyond $m=5$ the vertex problem has stood open since 1974.}

\tikzframesmall{The star hypertree}
  {\fighyperstar}
  {$r=3$, $n=9$: four hyperedges meeting only at the hub, Berge connectivity one, attaining $\floor{(n-1)/(r-1)}=4$. One colour is one hyperedge, a line stopping at all $r$ of its vertices and able to carry one route.}
  {0.55}

\begin{frame}{The hypergraph vertex problem}
\vfill
\centering
{\color{black!45}\small At $m=2$, for every $r\ge2$ and $n\ge1$:}\\[6pt]
{\color{KULblauw1}\Large $\displaystyle K_2^{(r)}(n)=\floorfrac{n-1}{r-1}$}\\[4pt]
{\footnotesize\color{black!55} vertex and edge agree at $m=2$ for every uniformity, and repeated hyperedges never help}

\vspace{10pt}
{\color{black!45}\small At $m=3$, a universal upper bound for every $r$-uniform multihypergraph:}\\[6pt]
{\color{KULblauw1}\Large $\displaystyle |E(\mathcal{H})|\le\floorfrac{2(n-1)}{r-1}$}\\[6pt]
{\footnotesize\color{black!55} attained by a \emph{simple} hypergraph for $r\ge3$ once $2\le\binom{n-2}{r-2}$, so $k_3^{(r)}(n)=\ell_3^{(r)}(n)$ there, and by a multihypergraph whenever $(r-1)\mid(n-1)$}

\vspace{8pt}
\begin{minipage}{0.86\textwidth}
{\footnotesize\color{black!60} Both read Berge routes as ordinary paths in the incidence graph. The $m=3$ case needs a new rank bound on that incidence graph, proved by induction over blocks and separating pairs. At $m\ge4$ the route stops for want of a $4$-connectivity analogue, and the value is open.}
\end{minipage}
\vfill
\end{frame}

\sectionframe{Part 4}{The directed turn}

\tikzframe{Two shapes tie at $n=7$}
  {\figdigraphmtwo}
  {The bidirected star holds $2(n-1)=12$ arcs, the one-directional wall $B_{3,4}$ holds $\floor{n^2/4}=12$, both at $\lec^{\max}=1$. Hence $\ell_2^{\dir}(n)=\max\bigl(2(n-1),\floor{n^2/4}\bigr)$, and $k_2^{\dir}(n)$ is the same value.}

\figframe{Linear hub against quadratic wall}
  {directed_crossover.png}
  {The hub spends its arcs on one centre and grows linearly. The augmented one-way wall grows quadratically and takes the lead near $n=9$ when $m=3$. No undirected graph can be this dense at a fixed connectivity ceiling.}

\tikzframe{The construction that carries the quadratic to every $m$}
  {\figaugmentedwall}
  {At $m=3$, $n=10$: every arc runs $A\to B$, and inside $B$ each vertex takes $m-2$ arcs from its cyclic predecessors. A pair $a_i,b_j$ then has the direct arc and one detour, so $\lec^{\max}=2$, and $24+6=30$ beats the hub's $27$.}

\begin{frame}{The directed conjecture, and what is unconditional}
\vfill
\centering
{\color{black!45}\small Conjectured value, lower bound proved for all $m$, exhausted at $n\le6$ when $m=3$:}\\[6pt]
{\color{stConj}\large $\displaystyle \ell_m^{\dir}(n)=\max\Bigl(m(n-1),\ \floorfrac{(n+m-2)^2}{4}\Bigr)$}

\vspace{18pt}
{\color{black!45}\small Proved here with no hypothesis on the extremal shape, for both separations:}\\[6pt]
{\color{KULblauw1}\large $\displaystyle |A(D)| \;\le\; \floorfrac{n^2}{4} \;+\; 4(m-1)(n-1)$}\\[8pt]
{\color{black!45}\small and therefore, for each fixed $m\ge3$,}\\[6pt]
{\color{KULblauw1}\large $\displaystyle \ell_m^{\dir}(n),\ k_m^{\dir}(n) \;=\; \frac{n^2}{4}+\Theta_m(n)$}

\vspace{14pt}
\begin{minipage}{0.88\textwidth}
{\footnotesize\color{black!60} The engine is a two-step route count: a vertex cannot have many out-neighbours that reach one another, or the short detours breach the ceiling. Leading constant and order of the second term are both theorems. What is left is one coefficient, $(m-2)/2$ conjectured against $4(m-1)$ proved.}
\end{minipage}
\vfill
\end{frame}

\begin{frame}{Shape as well as size}
\vfill
\centering
\begin{minipage}{0.88\textwidth}
{\color{KULblauw1}\bfseries In size.} {\small The extremal value lies within $O_m(n)$ of the wall's count, so what is open is a constant inside the second-order term rather than the phenomenon.}

\vspace{16pt}
{\color{KULblauw1}\bfseries In shape.} {\small Every feasible digraph becomes a subgraph of a one-directional bipartite graph after deleting $O(\sqrt{m}\,n^{3/2})$ arcs. A weaker error term, because it comes from the count that does not induct on a low-degree vertex.}
\end{minipage}

\vspace{24pt}
\begin{minipage}{0.88\textwidth}
{\footnotesize\color{black!60} What a full upper bound still needs: find the partition, rule out the arcs running back, and control the internal structure of the larger part $B$. One returning arc alone breaks the count, since it lets a route leave $B$ through $A$ and exceed the internal budget of $m-2$. Ruling that arc out for every extremal example is the missing step.}
\end{minipage}
\vfill
\end{frame}

\begin{frame}{One-way multigraphs, solved outright}
\vfill
\centering
{\color{KULblauw1}\Large $\displaystyle L_m^{\dir}(n)=(m-1)\,M(n), \qquad M(n)=\max\Bigl(2(n-1),\ \floorfrac{n^2}{4}\Bigr)$}\\[6pt]
{\color{black!45}\small for every $n\ge2$ and every $m\ge2$}

\vspace{22pt}
\begin{minipage}{0.88\textwidth}
{\small The proof deletes a \emph{subgraph} at a time rather than a vertex. Inside any feasible multigraph there is a sparse skeleton that alone preserves which vertices reach which, and removing one copy of each of its arcs costs every pair at least one route. Peeling off $m-1$ such skeletons accounts for every arc, and the parity obstruction that stopped the vertex-deletion induction never arises.}

\vspace{14pt}
{\footnotesize\color{black!60} No parity split and no auxiliary conjecture. Equality also classifies the quadratic extremisers once $n\ge\max(8,2m)$, while the linear-branch extremisers are exhibited rather than classified. The finite $n=7$ fact this problem once needed a solver for now follows in three lines.}
\end{minipage}
\vfill
\end{frame}

\begin{frame}{One-way hyperedge networks}
\vfill
\centering
{\color{black!45}\small For every forward directed $r$-uniform multihypergraph with $\kmax_{\mathcal{H}}\le m-1$:}\\[8pt]
{\color{KULblauw1}\Large $\displaystyle |E(\mathcal{H})| \;\le\; \frac{m-1}{r-1}\floorfrac{n^2}{4} \;+\; 4(m-1)^2(n-1)$}

\vspace{18pt}
{\color{black!45}\small so for fixed $r$ and $m$, under \emph{both} separations,}\\[6pt]
{\color{KULblauw1}\Large $\displaystyle \bigl(1+o(1)\bigr)\,\frac{(m-1)\,n^2}{4(r-1)}$}

\vspace{20pt}
\begin{minipage}{0.88\textwidth}
{\footnotesize\color{black!60} The proof forms the one-step shadow digraph, whose arc $u\to v$ records that some hyperedge has tail $u$ and head $v$. A matching argument transfers the route ceiling to the shadow's two-step budget, and a tail--head incidence count converts shadow arcs back to hyperedges. The bipartite construction already attains that leading term, so it is asymptotically optimal. Before this, the bounds were a factor of four apart.}
\end{minipage}
\vfill
\end{frame}

\begin{frame}{What each job is allowed to claim}
\vfill
\begin{columns}[T]
\begin{column}{0.32\textwidth}
  {\color{roleMeasure}\bfseries\large MEASURE}\\[8pt]
  {\small One integer maximum flow, on one multiplicity matrix or one hyperedge list, for all sixteen variants.}\\[8pt]
  {\scriptsize\color{black!55} What it reports about a given graph is a fact.}
\end{column}
\begin{column}{0.32\textwidth}
  {\color{roleProve}\bfseries\large PROVE}\\[8pt]
  {\small Exhaust every graph of a fixed size, pruned, or seeded by \code{geng}.}\\[8pt]
  {\scriptsize\color{black!55} A finished sweep is an upper bound at that size. It is a finite solver check, not a reusable certificate.}
\end{column}
\begin{column}{0.32\textwidth}
  {\color{roleDiscover}\bfseries\large DISCOVER}\\[8pt]
  {\small Climb with a tabu list, from the empty graph, restarted from many seeds.}\\[8pt]
  {\scriptsize\color{black!55} Only ever exhibits a graph. Lower bounds only. Never a proof.}
\end{column}
\end{columns}

\vspace{24pt}
\centering
{\color{black!55}\small The search proposes a lower bound, a proof or a finished exhaustion disposes the upper bound,}\\
{\color{black!55}\small and only when the two meet is a value called settled. No theorem here rests on an unchecked solver result.}
\vfill
\end{frame}

\tikzframe{One flow engine for sixteen variants}
  {\figvertexsplit}
  {Every vertex-disjoint question becomes an edge-disjoint one: split $v$ into $v^{\mathrm{in}}$ and $v^{\mathrm{out}}$ joined by a unit-capacity arc, so two routes through $v$ compete for that unit. Undirected edges become opposite arc pairs, and each hyperedge becomes a gate whose single internal arc carries its multiplicity.}

\begin{frame}{Proving a fixed size, three ways}
\vfill
\centering
{\small\input{../figures/generating_benchmark_table.tex}}

\vspace{14pt}
\begin{minipage}{0.88\textwidth}
{\footnotesize\color{black!60} Wall-clock time to prove $\ell_2^{\dir}(n)$ on one core. The blind sweep walks every matrix assignment, and it is dropped past $n=4$, where $2^{n(n-1)}$ assignments already cost $6.3$\,s and each further vertex multiplies that by $2^{2n}$. The pruned search stops a branch the moment a pair exceeds the ceiling, since adding arcs never lowers connectivity. The generated pipeline calls \code{geng} from \code{nauty} for one representative per isomorphism class and decorates its edges with arc multiplicities, inheriting the isomorphism reduction. From $n=5$ the gap widens every vertex: $68$ times faster at $n=7$. All three return the same proved value wherever they finish.}
\end{minipage}
\vfill
\end{frame}

\begin{frame}{Proving every $m$ at once}
\vfill
\centering
\begin{minipage}{0.86\textwidth}
{\large Divide every multiplicity of a feasible directed multigraph by $m-1$.}
\end{minipage}

\vspace{18pt}
{\color{KULblauw1}\large weights $w(u,v)\in[0,1]$, \quad pairwise maximum flow $\le1$,}\\[6pt]
{\color{KULblauw1}\large total weight $=$ arc count $/\,(m-1)$}

\vspace{22pt}
\begin{minipage}{0.88\textwidth}
{\footnotesize\color{black!60} One $m$-free optimisation therefore bounds every threshold at once, for a fixed $n$. The general theorem $L_m^{\dir}(n)=(m-1)M(n)$ has since replaced this route in the solver, but the older formulation is kept as an independent finite check, and the encoding is exact only up to $n=6$: at $n=7$ the certifier ran out of time without an answer.}
\end{minipage}
\vfill
\end{frame}

\sectionframe{Part 6}{Discovery}

\begin{frame}{One number per graph}
\vfill
\centering
{\color{KULblauw3b}\Large $\displaystyle \Phi(G) \;=\; -|E(G)| \;+\; \Lambda\cdot\max\bigl(0,\ c^{\max}(G)-(m-1)\bigr)$}

\vspace{8pt}
{\footnotesize\color{black!55} rewards edges, arcs or hyperedges, penalises any excess connectivity, with $c^{\max}$ standing for $\lmax$ or $\kmax$, whichever the run measures}

\vspace{22pt}
\begin{minipage}{0.88\textwidth}
{\small \textbf{Tabu search.} Start from the empty graph and toggle one adjacency at a time. Among the moves not forbidden, always take the one that lowers $\Phi$ most, \emph{even when every candidate raises it}. The pair just changed goes on a tabu list for a fixed number of steps, so the walk cannot immediately undo itself and cycle back. A move that would beat the best graph seen overrides the tabu flag.}

\vspace{12pt}
{\footnotesize\color{black!60} Why the walk is allowed above the ceiling: a feasible graph can be locally maximal and still far below the densest one. Improving it needs an add and then a different removal, a swap whose first half is infeasible. Going above the bound is never the goal. It is the shortcut on the way back down to it. The checker is also skipped whenever monotonicity already fixes the answer, so a flow is computed only for an addition made \emph{at} the ceiling.}
\end{minipage}
\vfill
\end{frame}

\begin{frame}{Tabu against annealing}
\vfill
\centering
\renewcommand{\arraystretch}{1.5}
\small
\begin{tabular}{@{}llcrr@{}}
\toprule
Variant & & target & tabu & annealing \\
\midrule
directed multigraph & $n=7$, $m=3$ & $24$ & $\mathbf{24}$ & $20$ \\
simple digraph      & $n=7$, $m=3$ & $18$ & $\mathbf{18}$ & $16$ \\
\bottomrule
\end{tabular}

\vspace{20pt}
\begin{minipage}{0.88\textwidth}
{\footnotesize\color{black!60} Simulated annealing runs on the same energy and the same neighbourhood, accepting a worsening move by the Metropolis rule and cooling as it goes. At equal wall-clock budget it is the weaker engine here: tabu takes fewer, better-aimed steps and assembles the structured extremiser the cooled walk cannot reach. Every discovery value the thesis reports therefore comes from tabu search, and annealing is kept only as an independent cross-check in the self-test, where agreement between two unrelated walks is worth more than the speed of either.}
\end{minipage}
\vfill
\end{frame}

\begin{frame}{What is new here}
\vfill
\small
\begin{itemize}\setlength{\itemsep}{4pt}
  \item[\color{KULblauw1}\textbf{1}] \textbf{One-way multigraphs, solved outright: $L_m^{\dir}(n)=(m-1)\max\bigl(2(n-1),\floor{n^2/4}\bigr)$}\\
        {\footnotesize\color{black!60} every $n$ and every $m$, by peeling off one sparse reachability-preserving subgraph per unit of $m-1$, with the quadratic-branch extremisers classified for $n\ge\max(8,2m)$.}
  \item[\color{KULblauw1}\textbf{2}] \textbf{One-way simple graphs: an unconditional $n^2/4+\Theta_m(n)$, arc and vertex alike}\\
        {\footnotesize\color{black!60} no assumption about the extremal shape, plus an $O(\sqrt{m}\,n^{3/2})$ structural distance from a one-way bipartite graph. The vertex statement is not a corollary of the arc one.}
  \item[\color{KULblauw1}\textbf{3}] \textbf{The hypergraph vertex problem: exact at $m=2$, and sharp at $m=3$ in its construction range}\\
        {\footnotesize\color{black!60} a universal $m=3$ upper bound from a new incidence-graph rank lemma, with equality for $r\ge3$ once $2\le\binom{n-2}{r-2}$.}
  \item[\color{KULblauw1}\textbf{4}] \textbf{Multigraph vertex routes with parallel copies counted as distinct}\\
        {\footnotesize\color{black!60} exact at $m=2$, counterexamples to the thickened-tree guess for every $m\ge5$, matching order $\Theta(m^2n)$, and an exact block-knapsack formulation.}
  \item[\color{KULblauw1}\textbf{5}] \textbf{One-way hyperedge networks: the leading term settled}\\
        {\footnotesize\color{black!60} $(1+o(1))(m-1)n^2/\bigl(4(r-1)\bigr)$ for fixed $r,m$, under both separations and every orientation convention, so the bipartite construction is asymptotically optimal.}
\end{itemize}
\vfill
\end{frame}

\begin{frame}{Orientations of a directed hyperedge}
\vfill
\centering
\begin{minipage}{0.86\textwidth}
{\small The tables use the \textbf{forward} convention: one tail and $r-1$ heads. Reversing every hyperedge makes the one-head \textbf{backward} convention equivalent to it. And the same leading term extends to \emph{every} other way of splitting a hyperedge into tail and head sets, under both separations.}
\end{minipage}

\vspace{22pt}
{\color{KULblauw3b}\large The orientation therefore adds no asymptotic row.}

\vspace{22pt}
\begin{minipage}{0.86\textwidth}
{\footnotesize\color{black!60} The conventions may still differ at few vertices, where $n\approx r$. Whether they agree exactly once $n>r$ is open, so Problem 915 stays a sixteen-variant question.}
\end{minipage}
\vfill
\end{frame}

\begin{frame}{Limits of the program}
\vfill
\begin{itemize}\setlength{\itemsep}{10pt}
  \item[{\color{stOpen}$\blacksquare$}] The search finds examples. It never proves an upper bound.
  \item[{\color{stOpen}$\blacksquare$}] The certifier proves an upper bound at one fixed size, as a finite solver check rather than a replayable certificate.
  \item[{\color{stOpen}$\blacksquare$}] The directed multigraph encoding is exact only to $n=6$. At $n=7$ the certifier ran out of time, and the value comes from the hand proof instead.
  \item[{\color{stOpen}$\blacksquare$}] The $n=7$ generation classification is verified only for extremisers of maximum total degree at most $8$.
\end{itemize}

\vspace{18pt}
\begin{minipage}{0.9\textwidth}
{\footnotesize\color{black!60} Three of the remaining gaps cannot be closed by computation at all: the decomposition behind the directed conjecture, the directed vertex problem at $m\ge3$, and the undirected vertex problem at $m\ge6$. Every computational claim can be rerun from \code{program/erdos915\_unified.py}, and the important values are checked by independent implementations.}
\end{minipage}
\vfill
\end{frame}
